# /etc/systemd/system/exocomp-coordinator.service
[Unit]
Description=Exocomp Coordinator
Documentation=https://github.com/NVShawn/exocomp
After=network-online.target
Wants=network-online.target

# Bound automatic restart churn. Start-limit directives belong to this unit
# section; systemd ignores them when placed in the service section.
StartLimitIntervalSec=120s
StartLimitBurst=4

# ── Ordering and dependency ────────────────────────────────────────────────────
# Require network before starting.  Node connections are accepted after start.

[Service]
Type=exec

# ── Identity ──────────────────────────────────────────────────────────────────
# Run as a dedicated unprivileged account created by the installer.
# exocomp-coordinator and /opt/exocomp/coordinator are substituted by install.sh at install time.
User=exocomp-coordinator
Group=exocomp-coordinator

# ── Executable ────────────────────────────────────────────────────────────────
ExecStart=/opt/exocomp/coordinator/current/bin/exocomp_coordinator start
# The release stop command requires distributed Erlang RPC, which may not be
# ready during a failed upgrade. The BEAM handles SIGTERM gracefully.
ExecStop=/bin/kill -TERM $MAINPID
ExecReload=/bin/kill -HUP $MAINPID

WorkingDirectory=/opt/exocomp/coordinator

# ── Environment ───────────────────────────────────────────────────────────────
Environment=EXOCOMP_COORDINATOR_CONFIG_FILE=/opt/exocomp/coordinator/config/coordinator.json
Environment=EXOCOMP_PKI_ONLINE_STATE=/var/lib/exocomp-coordinator/pki
Environment=EXOCOMP_ENROLLMENT_TOKEN_STORE=/var/lib/exocomp-coordinator/enrollment-tokens
Environment=EXOCOMP_TLS_CA_PATH=/var/lib/exocomp-coordinator/pki/root_ca.pem
Environment=EXOCOMP_TLS_CERT_PATH=/var/lib/exocomp-coordinator/pki/coordinator.pem
Environment=EXOCOMP_A2A_TLS_CERT_PATH=/var/lib/exocomp-coordinator/pki/coordinator_chain.pem
Environment=EXOCOMP_TLS_KEY_PATH=/var/lib/exocomp-coordinator/pki/coordinator_key.pem
Environment=EXOCOMP_LOG_DIR=/opt/exocomp/coordinator/log
Environment=ERL_CRASH_DUMP=/opt/exocomp/coordinator/log/erl_crash.dump
EnvironmentFile=-/opt/exocomp/coordinator/config/release-cookie.env
Environment=HOME=/opt/exocomp/coordinator

# ── Runtime directories ───────────────────────────────────────────────────────
RuntimeDirectory=exocomp-coordinator
RuntimeDirectoryMode=0750

StateDirectory=exocomp-coordinator
StateDirectoryMode=0750

# ── Restart policy ───────────────────────────────────────────────────────────
Restart=on-failure
RestartSec=5s
TimeoutStartSec=60s
TimeoutStopSec=60s

# ── Logging ───────────────────────────────────────────────────────────────────
StandardOutput=journal
StandardError=journal
SyslogIdentifier=exocomp-coordinator

# ── Capability hardening ──────────────────────────────────────────────────────
CapabilityBoundingSet=
AmbientCapabilities=

# ── Privilege escalation prevention ──────────────────────────────────────────
NoNewPrivileges=true
SecureBits=noroot noroot-locked

# ── Filesystem access ─────────────────────────────────────────────────────────
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/opt/exocomp/coordinator/log /opt/exocomp/coordinator/config /var/lib/exocomp-coordinator

PrivateTmp=true

ProtectKernelTunables=true
ProtectKernelModules=true
ProtectKernelLogs=true
ProtectControlGroups=true

RestrictNamespaces=true
ProtectClock=true
ProtectHostname=true
RestrictSUIDSGID=true

# ── Process and memory restrictions ──────────────────────────────────────────
MemoryDenyWriteExecute=true
LockPersonality=true
RestrictRealtime=true

# ── Device access ─────────────────────────────────────────────────────────────
PrivateDevices=true

# ── System call filter ────────────────────────────────────────────────────────
# The coordinator manages PKI, accepts TLS connections from nodes, and talks
# to an offline root CA.  The same BEAM service syscall profile applies.
SystemCallFilter=@system-service @network-io @file-system @process
SystemCallFilter=~@clock @cpu-emulation @debug @keyring @module @mount @raw-io @reboot @resources @swap @obsolete
SystemCallErrorNumber=EPERM

# ── Address family restriction ────────────────────────────────────────────────
RestrictAddressFamilies=AF_INET AF_INET6 AF_UNIX

# ── Resource limits ───────────────────────────────────────────────────────────
LimitNOFILE=65536
LimitCORE=0

# ── umask ─────────────────────────────────────────────────────────────────────
UMask=0077

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/exocomp-node.service
[Unit]
Description=Exocomp Node Agent
Documentation=https://github.com/NVShawn/exocomp
After=network-online.target
Wants=network-online.target

# Bound automatic restart churn. Start-limit directives belong to this unit
# section; systemd ignores them when placed in the service section.
StartLimitIntervalSec=120s
StartLimitBurst=4

# ── Ordering and dependency ────────────────────────────────────────────────────
# Require network connectivity before starting.  The coordinator connection is
# established after start, not as a hard systemd dependency, because the node
# retries the connection internally with back-off.

[Service]
Type=exec

# ── Identity ──────────────────────────────────────────────────────────────────
# Run as a dedicated unprivileged account created by the installer.
# exocomp-node and /opt/exocomp/node are substituted by install.sh at install time.
User=exocomp-node
Group=exocomp-node

# ── Executable ────────────────────────────────────────────────────────────────
ExecStart=/opt/exocomp/node/current/bin/exocomp_node start
# The release stop command requires distributed Erlang RPC, which may not be
# ready during a failed upgrade. The BEAM handles SIGTERM gracefully.
ExecStop=/bin/kill -TERM $MAINPID
ExecReload=/bin/kill -HUP $MAINPID

WorkingDirectory=/opt/exocomp/node

# ── Environment ───────────────────────────────────────────────────────────────
Environment=EXOCOMP_CONFIG_FILE=/opt/exocomp/node/config/node.json
Environment=EXOCOMP_LOG_DIR=/opt/exocomp/node/log
Environment=ERL_CRASH_DUMP=/opt/exocomp/node/log/erl_crash.dump
EnvironmentFile=-/opt/exocomp/node/config/release-cookie.env
# Prevent BEAM from trying to write to ${HOME}
Environment=HOME=/opt/exocomp/node

# ── Runtime directories ───────────────────────────────────────────────────────
# systemd creates and removes these automatically.
RuntimeDirectory=exocomp-node
RuntimeDirectoryMode=0750

# State directory for persistent data (kept across restarts and upgrades).
StateDirectory=exocomp-node
StateDirectoryMode=0750

# Log directory managed by the installer (not systemd) so audit logs survive
# across package upgrades and uninstalls.  Installer creates it with correct
# ownership.

# ── Restart policy ───────────────────────────────────────────────────────────
Restart=on-failure
RestartSec=5s
TimeoutStartSec=30s
TimeoutStopSec=30s

# ── Logging ───────────────────────────────────────────────────────────────────
StandardOutput=journal
StandardError=journal
SyslogIdentifier=exocomp-node

# ── Capability hardening ──────────────────────────────────────────────────────
# The service itself starts without effective or ambient capabilities.  Keep
# only setuid/setgid in the bounding set so the fixed /usr/bin/sudo actions can
# cross to the exact root commands installed in /etc/sudoers.d/exocomp-node.
CapabilityBoundingSet=CAP_SETGID CAP_SETUID
AmbientCapabilities=

# ── Privilege escalation prevention ──────────────────────────────────────────
# NoNewPrivileges and SecureBits=noroot would make the exact sudo action
# catalog impossible to execute.  Privilege remains bounded by the generated
# argument-exact NOPASSWD policy; no generic sudo or systemctl grant exists.
NoNewPrivileges=false

# ── Filesystem access ─────────────────────────────────────────────────────────
# Mount the system namespace read-only and grant write access only to the
# directories the service legitimately needs.
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/opt/exocomp/node/log /opt/exocomp/node/config /var/lib/exocomp-node

# Private /tmp prevents the service from accessing /tmp state written by other
# processes.
PrivateTmp=true

# The service does not use kernel debugging, modules, or control groups.
ProtectKernelTunables=true
ProtectKernelModules=true
ProtectKernelLogs=true
ProtectControlGroups=true

# Prevent namespace creation and clock changes.
RestrictNamespaces=true
ProtectClock=true

# Do not expose host identity beyond what is needed.
ProtectHostname=true

# Block setuid/setgid bits on executed files.
RestrictSUIDSGID=true

# ── Process and memory restrictions ──────────────────────────────────────────
# Deny mmap with PROT_EXEC | PROT_WRITE (code injection mitigation).
MemoryDenyWriteExecute=true

# Lock the execution domain / personality.
LockPersonality=true

# Do not allow real-time scheduling (prevents starvation attacks).
RestrictRealtime=true

# ── Device access ─────────────────────────────────────────────────────────────
# Mount a private /dev with only the handful of pseudo-devices that are always
# safe (null, random, urandom, tty).
PrivateDevices=true

# ── System call filter ────────────────────────────────────────────────────────
# Allow only the syscalls needed by a typical BEAM-based service.
# @system-service covers the minimal set for init/supervision; remaining BEAM
# syscalls (futex, io_uring compatible subset, epoll, etc.) are included.
SystemCallFilter=@system-service @network-io @file-system @process
SystemCallFilter=~@clock @cpu-emulation @debug @keyring @module @mount @raw-io @reboot @resources @swap @obsolete
SystemCallErrorNumber=EPERM

# ── Address family restriction ────────────────────────────────────────────────
# The node connects to the coordinator over TCP/TLS and binds a local health
# socket.  UNIX sockets are needed for gRPC and internal communication.
RestrictAddressFamilies=AF_INET AF_INET6 AF_UNIX

# ── Resource limits ───────────────────────────────────────────────────────────
# Limit file descriptors (BEAM default is high; cap to a reasonable amount).
LimitNOFILE=65536
# Limit core dumps (suppress by default; set to unlimited for debugging).
LimitCORE=0

# ── umask ─────────────────────────────────────────────────────────────────────
# Files created by the service are readable only by the service account.
UMask=0077

[Install]
WantedBy=multi-user.target
